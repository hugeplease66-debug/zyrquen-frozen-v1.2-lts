"use client";

import React, { useState, useMemo } from "react";
import { 
  LineChart, 
  Line, 
  XAxis, 
  YAxis, 
  Tooltip, 
  ResponsiveContainer,
  AreaChart,
  Area,
  ReferenceLine
} from "recharts";
import { 
  Activity, 
  AlertTriangle, 
  CheckCircle2, 
  Flame, 
  Snowflake, 
  Zap, 
  Cpu, 
  Radio, 
  RefreshCw, 
  Sliders, 
  ShieldAlert, 
  Clock, 
  Sparkles 
} from "lucide-react";
import { SSOT } from "@/lib/ssot-data";

export interface TelemetryPredictionPoint {
  timeOffset: number; // in seconds: 0 to +60
  actualQOps?: number;
  predictedQOps: number;
  predictedCryoTemp: number; // mK
  predictedCoreTemp: number; // deg C
  predictedBandwidth: number; // GB/s
  anomalyRisk: number; // 0 - 100%
}

export const QuantumAnomalyPredictor: React.FC = () => {
  const [forecastHorizon, setForecastHorizon] = useState<number>(60); // 10s to 60s
  const [scrubbedTime, setScrubbedTime] = useState<number>(30); // scrub time
  const [anomalyMode, setAnomalyMode] = useState<"NOMINAL" | "THERMAL_SPIKE" | "QOPS_COLLAPSE" | "BANDWIDTH_DECAY">("NOMINAL");
  const [isCalibrating, setIsCalibrating] = useState<boolean>(false);

  // Generate predictive 60-second curve based on mode
  const predictionData = useMemo<TelemetryPredictionPoint[]>(() => {
    const points: TelemetryPredictionPoint[] = [];
    for (let t = 0; t <= 60; t += 5) {
      let qops = 851.92;
      let cryo = 14.98;
      let core = 42.5;
      let bw = 19.8;
      let risk = 0.5;

      if (anomalyMode === "THERMAL_SPIKE") {
        // Core temp rises towards 85C, cryo fluctuates, risk surges
        const factor = t / 60;
        core = +(42.5 + factor * 44.0).toFixed(1); // reaches 86.5 C (Breaches 85C threshold!)
        cryo = +(14.98 + factor * 0.12).toFixed(3);
        qops = +(851.92 - factor * 120).toFixed(2);
        bw = +(19.8 - factor * 4.2).toFixed(1);
        risk = +(factor * 94).toFixed(1);
      } else if (anomalyMode === "QOPS_COLLAPSE") {
        const factor = t / 60;
        qops = +(851.92 - factor * 450).toFixed(2);
        core = +(42.5 + factor * 15).toFixed(1);
        cryo = +(14.98 + factor * 0.05).toFixed(3);
        bw = +(19.8 - factor * 6.5).toFixed(1); // drops to 13.3 GB/s (< 15GB/s Fail Closed!)
        risk = +(factor * 88).toFixed(1);
      } else if (anomalyMode === "BANDWIDTH_DECAY") {
        const factor = t / 60;
        bw = +(19.8 - factor * 8.0).toFixed(1); // drops to 11.8 GB/s
        qops = +(851.92 - factor * 200).toFixed(2);
        core = +(42.5 + factor * 8).toFixed(1);
        risk = +(factor * 82).toFixed(1);
      } else {
        // Nominal slight quantum jitter
        const noise = (Math.sin(t / 5) * 2.5);
        qops = +(851.92 + noise).toFixed(2);
        cryo = +(14.98 + Math.cos(t / 8) * 0.008).toFixed(3);
        core = +(42.5 + Math.sin(t / 10) * 0.8).toFixed(1);
        bw = +(19.8 + Math.cos(t / 6) * 0.2).toFixed(1);
        risk = 1.2;
      }

      points.push({
        timeOffset: t,
        actualQOps: t <= 10 ? +(qops + Math.sin(t * 1.7) * 2).toFixed(2) : undefined,
        predictedQOps: qops,
        predictedCryoTemp: cryo,
        predictedCoreTemp: core,
        predictedBandwidth: bw,
        anomalyRisk: risk
      });
    }
    return points;
  }, [anomalyMode]);

  // Scrubbed point
  const currentScrubbedPoint = predictionData.find(p => p.timeOffset === scrubbedTime) || predictionData[6] || {
    timeOffset: 30,
    predictedQOps: 851.92,
    predictedCryoTemp: 14.98,
    predictedCoreTemp: 42.5,
    predictedBandwidth: 19.8,
    anomalyRisk: 1.2
  };

  // Check Fail-Closed automatic thresholds
  const isCoreOverheat = currentScrubbedPoint.predictedCoreTemp > SSOT.failClosedThresholds.maxCoreTemp;
  const isBandwidthLow = currentScrubbedPoint.predictedBandwidth < SSOT.failClosedThresholds.minBandwidth;
  const isFailClosedTriggered = isCoreOverheat || isBandwidthLow;

  const handleNeutralize = () => {
    setIsCalibrating(true);
    setTimeout(() => {
      setAnomalyMode("NOMINAL");
      setIsCalibrating(false);
    }, 800);
  };

  return (
    <div className="bg-zinc-950 border border-cyan-500/50 rounded-3xl p-5 sm:p-6 shadow-2xl space-y-6 text-zinc-100">
      {/* Header */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4 border-b border-zinc-800 pb-4">
        <div className="flex items-center gap-3">
          <div className="w-10 h-10 rounded-2xl bg-cyan-500/15 border border-cyan-500/60 flex items-center justify-center text-cyan-400">
            <Radio className="w-5 h-5" />
          </div>
          <div>
            <div className="flex items-center gap-2">
              <span className="text-[10px] font-mono font-bold uppercase tracking-wider text-cyan-400">
                AI TELEMETRY TIME-SERIES FORECASTING
              </span>
              <span className="px-2 py-0.5 rounded-full bg-cyan-950 text-cyan-300 border border-cyan-700/60 text-[10px] font-mono font-bold">
                T+60 SECONDS HORIZON
              </span>
            </div>
            <h2 className="text-base sm:text-lg font-bold font-mono text-zinc-100 mt-0.5">
              Quantum Anomaly Predictor (Thermodynamic &amp; QOps Stability Engine)
            </h2>
          </div>
        </div>

        {/* Anomaly Stress Test Buttons */}
        <div className="flex items-center gap-2 flex-wrap">
          <span className="text-[11px] font-mono text-zinc-400">Inject Vector:</span>
          <button
            onClick={() => setAnomalyMode("NOMINAL")}
            className={`px-2.5 py-1.5 rounded-xl font-mono text-xs font-bold transition-all cursor-pointer ${anomalyMode === "NOMINAL" ? "bg-cyan-950 text-cyan-300 border border-cyan-500" : "bg-zinc-900 text-zinc-400 border border-zinc-800"}`}
          >
            Nominal
          </button>
          <button
            onClick={() => setAnomalyMode("THERMAL_SPIKE")}
            className={`px-2.5 py-1.5 rounded-xl font-mono text-xs font-bold transition-all cursor-pointer ${anomalyMode === "THERMAL_SPIKE" ? "bg-rose-950 text-rose-300 border border-rose-600" : "bg-zinc-900 text-zinc-400 border border-zinc-800"}`}
          >
            Thermal Spike (&gt;85°C)
          </button>
          <button
            onClick={() => setAnomalyMode("QOPS_COLLAPSE")}
            className={`px-2.5 py-1.5 rounded-xl font-mono text-xs font-bold transition-all cursor-pointer ${anomalyMode === "QOPS_COLLAPSE" ? "bg-amber-950 text-amber-300 border border-amber-600" : "bg-zinc-900 text-zinc-400 border border-zinc-800"}`}
          >
            QOps Collapse
          </button>
          <button
            onClick={() => setAnomalyMode("BANDWIDTH_DECAY")}
            className={`px-2.5 py-1.5 rounded-xl font-mono text-xs font-bold transition-all cursor-pointer ${anomalyMode === "BANDWIDTH_DECAY" ? "bg-purple-950 text-purple-300 border border-purple-600" : "bg-zinc-900 text-zinc-400 border border-zinc-800"}`}
          >
            BW Decay (&lt;15GB/s)
          </button>
        </div>
      </div>

      {/* 60s Future Scrubbing Slider */}
      <div className="p-4 bg-black/70 border border-zinc-800 rounded-2xl space-y-3 font-mono">
        <div className="flex items-center justify-between text-xs">
          <span className="text-zinc-300 font-bold flex items-center gap-1.5">
            <Clock className="w-4 h-4 text-cyan-400" />
            Forecast Horizon Timeline Scrub: T+{scrubbedTime}s into the future
          </span>
          <span className={`text-[11px] px-2 py-0.5 rounded-full font-bold ${
            isFailClosedTriggered
              ? "bg-rose-950 text-rose-300 border border-rose-700 animate-pulse"
              : currentScrubbedPoint.anomalyRisk > 30
                ? "bg-amber-950 text-amber-300 border border-amber-700"
                : "bg-emerald-950 text-emerald-300 border border-emerald-700"
          }`}>
            {isFailClosedTriggered 
              ? "FAIL-CLOSED SENTRY TRIPPED" 
              : currentScrubbedPoint.anomalyRisk > 30 
                ? "ELEVATED ANOMALY RISK" 
                : "NOMINAL PREDICTIVE ENVELOPE"}
          </span>
        </div>

        <input
          type="range"
          min={0}
          max={60}
          step={5}
          value={scrubbedTime}
          onChange={(e) => setScrubbedTime(Number(e.target.value))}
          className="w-full h-2 bg-zinc-800 rounded-lg appearance-none cursor-pointer accent-cyan-400"
        />

        <div className="flex justify-between text-[10px] text-zinc-500">
          <span>T+0s (Present Telemetry)</span>
          <span>T+15s</span>
          <span>T+30s (Mid Horizon)</span>
          <span>T+45s</span>
          <span>T+60s (Deep Forecast)</span>
        </div>
      </div>

      {/* Predictive State Cards at Scrubbed Point */}
      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 font-mono text-xs">
        {/* Metric 1: Forecast QOps */}
        <div className="p-3.5 bg-black/60 border border-zinc-800 rounded-xl space-y-1">
          <span className="text-zinc-500 text-[10px] block">PREDICTED ENERGY DISPATCH</span>
          <div className="text-cyan-300 font-bold text-base flex items-center gap-1.5">
            <Zap className="w-4 h-4 text-cyan-400" />
            <span>{currentScrubbedPoint.predictedQOps} QOps/s</span>
          </div>
          <span className="text-[10px] text-zinc-400">Baseline: 851.92 QOps/s</span>
        </div>

        {/* Metric 2: Forecast Core Temp */}
        <div className={`p-3.5 rounded-xl border space-y-1 ${isCoreOverheat ? "bg-rose-950/40 border-rose-700 text-rose-300" : "bg-black/60 border-zinc-800"}`}>
          <span className="text-zinc-500 text-[10px] block">PREDICTED CORE TEMP</span>
          <div className={`font-bold text-base flex items-center gap-1.5 ${isCoreOverheat ? "text-rose-400" : "text-amber-400"}`}>
            <Flame className="w-4 h-4" />
            <span>{currentScrubbedPoint.predictedCoreTemp}°C</span>
          </div>
          <span className={`text-[10px] ${isCoreOverheat ? "text-rose-400 font-bold" : "text-zinc-400"}`}>
            {isCoreOverheat ? "BREACH: Limit 85.0°C" : "Safe Margin (<85°C)"}
          </span>
        </div>

        {/* Metric 3: Forecast Bandwidth */}
        <div className={`p-3.5 rounded-xl border space-y-1 ${isBandwidthLow ? "bg-rose-950/40 border-rose-700 text-rose-300" : "bg-black/60 border-zinc-800"}`}>
          <span className="text-zinc-500 text-[10px] block">PREDICTED MEMORY BUS</span>
          <div className={`font-bold text-base flex items-center gap-1.5 ${isBandwidthLow ? "text-rose-400" : "text-purple-400"}`}>
            <Cpu className="w-4 h-4" />
            <span>{currentScrubbedPoint.predictedBandwidth} GB/s</span>
          </div>
          <span className={`text-[10px] ${isBandwidthLow ? "text-rose-400 font-bold" : "text-zinc-400"}`}>
            {isBandwidthLow ? "BREACH: Min 15.0 GB/s" : "Safe Bus (>=15 GB/s)"}
          </span>
        </div>

        {/* Metric 4: Forecast Cryo Temp */}
        <div className="p-3.5 bg-black/60 border border-zinc-800 rounded-xl space-y-1">
          <span className="text-zinc-500 text-[10px] block">PREDICTED HE-4 CRYO</span>
          <div className="text-emerald-400 font-bold text-base flex items-center gap-1.5">
            <Snowflake className="w-4 h-4 text-emerald-400" />
            <span>{currentScrubbedPoint.predictedCryoTemp} mK</span>
          </div>
          <span className="text-[10px] text-zinc-400">Subzero Baseline 14.98 mK</span>
        </div>
      </div>

      {/* Recharts Predictive Chart: Actual vs Forecast 60s Ahead */}
      <div className="bg-black/80 border border-zinc-800 rounded-2xl p-4 space-y-3 font-mono">
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 text-xs">
          <div className="flex items-center gap-2">
            <Activity className="w-4 h-4 text-cyan-400" />
            <span className="text-zinc-200 font-bold">
              60-Second Predictive Telemetry Trajectory (QOps &amp; Anomaly Probability)
            </span>
          </div>
          <div className="flex items-center gap-4 text-[11px]">
            <span className="flex items-center gap-1.5 text-cyan-400">
              <span className="w-3 h-0.5 bg-cyan-400"></span>
              Predicted QOps
            </span>
            <span className="flex items-center gap-1.5 text-rose-400">
              <span className="w-3 h-0.5 bg-rose-500 border-dashed"></span>
              Anomaly Risk %
            </span>
          </div>
        </div>

        <div className="w-full h-56">
          <ResponsiveContainer width="100%" height="100%">
            <AreaChart data={predictionData}>
              <defs>
                <linearGradient id="riskGradient" x1="0" y1="0" x2="0" y2="1">
                  <stop offset="5%" stopColor="#f43f5e" stopOpacity={0.4}/>
                  <stop offset="95%" stopColor="#f43f5e" stopOpacity={0.0}/>
                </linearGradient>
              </defs>
              <XAxis 
                dataKey="timeOffset" 
                tickFormatter={(val) => `T+${val}s`} 
                stroke="#52525b"
                tick={{ fontSize: 10 }}
              />
              <YAxis 
                yAxisId="qops"
                domain={[300, 950]} 
                stroke="#06b6d4" 
                tick={{ fontSize: 10 }}
                orientation="left"
              />
              <YAxis 
                yAxisId="risk"
                domain={[0, 100]} 
                stroke="#f43f5e" 
                tick={{ fontSize: 10 }}
                orientation="right"
              />
              <Tooltip 
                contentStyle={{ 
                  backgroundColor: '#09090b', 
                  borderColor: '#27272a', 
                  borderRadius: '0.75rem',
                  fontSize: '11px',
                  fontFamily: 'monospace'
                }}
              />
              <ReferenceLine x={scrubbedTime} stroke="#eab308" strokeDasharray="3 3" label={{ value: `T+${scrubbedTime}s`, fill: '#eab308', fontSize: 10 }} />
              <Area 
                yAxisId="risk"
                type="monotone" 
                dataKey="anomalyRisk" 
                stroke="#f43f5e" 
                fill="url(#riskGradient)" 
                strokeWidth={2}
                name="Anomaly Risk (%)"
              />
              <Line 
                yAxisId="qops"
                type="monotone" 
                dataKey="predictedQOps" 
                stroke="#06b6d4" 
                strokeWidth={2.5} 
                dot={{ r: 2, fill: '#06b6d4' }}
                name="Forecast QOps"
              />
            </AreaChart>
          </ResponsiveContainer>
        </div>
      </div>

      {/* Proactive Countermeasure Action Bar */}
      <div className="p-4 bg-black/60 border border-zinc-800 rounded-2xl flex flex-col sm:flex-row sm:items-center justify-between gap-4 font-mono text-xs">
        <div className="space-y-1">
          <div className="text-zinc-300 font-bold flex items-center gap-2">
            <ShieldAlert className="w-4 h-4 text-amber-400" />
            <span>Automated SSoT Δ0 Proactive Stabilization Policy</span>
          </div>
          <p className="text-[11px] text-zinc-400">
            {isFailClosedTriggered 
              ? "คำเตือน: โมเดลพยากรณ์พบว่าระบบจะละเมิดเงื่อนไข Fail-Closed (Core Temp > 85°C หรือ BW < 15GB/s) ในอนาคต"
              : "ระบบอยู่ในเกณฑ์เสถียรภาพปกติ ไม่พบแนวโน้มการสูญเสียสภาพความสั่นพ้องควอนตัม"}
          </p>
        </div>

        <button
          onClick={handleNeutralize}
          disabled={isCalibrating}
          className="px-4 py-2 bg-gradient-to-r from-cyan-600 to-blue-600 hover:from-cyan-700 hover:to-blue-700 text-white font-bold rounded-xl flex items-center gap-2 cursor-pointer transition-all disabled:opacity-50 whitespace-nowrap shadow-md"
        >
          <RefreshCw className={`w-3.5 h-3.5 ${isCalibrating ? "animate-spin" : ""}`} />
          <span>{isCalibrating ? "Neutralizing Entropy..." : "Apply Proactive Cryo-Stabilization"}</span>
        </button>
      </div>
    </div>
  );
};

export default QuantumAnomalyPredictor;
