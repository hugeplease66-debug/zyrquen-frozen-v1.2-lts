            createdAt: serverTimestamp()

        setEvents(INITIAL_ANOMALIES);
      } else {
        const fetched: AnomalyEvent[] = [];
        snapshot.forEach((doc) => {
          fetched.push({ id: doc.id, ...doc.data() } as AnomalyEvent);
        });
        setEvents(fetched);
      }
      setLoading(false);
    }, (err) => {
      console.warn("Sentinel realtime fallback:", err);
      setEvents(INITIAL_ANOMALIES);
      setLoading(false);

    return () => unsubscribe();

            onClick={() => setFilterType('APPROVED')}
            className={`px-2.5 py-1 rounded-lg text-[11px] transition-colors ${
              filterType === 'APPROVED' ? 'bg-emerald-950/80 text-emerald-300 border border-emerald-700/60 font-bold' : 'text-zinc-400 hover:text-emerald-300'
            }`}
          >
            Approved ({approvedCount})
          </button>
        </div>

        <div className="text-[11px] text-zinc-500 flex items-center gap-2">
          <span>ETDA Act B.E. 2544 (Sec 9, 26, 28)</span>
          <span className="w-1.5 h-1.5 rounded-full bg-emerald-400"></span>
        </div>
      </div>

      {/* Realtime Anomaly Events Table / List */}
      <div className="divide-y divide-zinc-800/80 max-h-[460px] overflow-y-auto font-mono text-xs">
        {filteredEvents.map((ev) => {
          const isExpanded = expandedId === ev.id;
          const isCritical = ev.riskScore >= 0.85;

          return (
            <div 
              key={ev.id} 
              className={`p-3.5 transition-colors ${
                isCritical 
                  ? 'bg-rose-950/15 hover:bg-rose-950/25' 
                  : 'bg-zinc-950/40 hover:bg-zinc-900/40'
              }`}
            >
              <div 
                onClick={() => setExpandedId(isExpanded ? null : ev.id)}
                className="flex items-center justify-between cursor-pointer gap-3"
              >
                {/* Left: ID & Actor Name */}
                <div className="flex items-center gap-3 min-w-0">
                  <div className={`w-2.5 h-2.5 rounded-full flex-shrink-0 ${
                    isCritical 
                      ? 'bg-rose-500 animate-pulse' 
                      : ev.riskScore >= 0.3 
                      ? 'bg-amber-400' 
                      : 'bg-emerald-400'
                  }`}></div>
                  
                  <div className="min-w-0">
                    <div className="flex items-center gap-2">
                      <span className="text-zinc-200 font-bold font-mono text-xs truncate">
                        {ev.name}
                      </span>
                      <span className="text-[10px] text-zinc-500 font-mono">
                        ({ev.requestId})
                      </span>
                    </div>
                    <div className="text-[11px] text-zinc-400 font-sans truncate">
                      Segment: <span className="text-zinc-300 font-mono">{ev.targetSegment}</span> • Protocol: <span className={`font-mono ${
                        ev.cryptoScheme.includes('Dilithium') || ev.cryptoScheme.includes('SPHINCS') ? 'text-purple-300 font-bold' : 'text-zinc-400'
                      }`}>{ev.cryptoScheme}</span>
                    </div>
                  </div>
                </div>

                {/* Right: Risk Score Gauge, Decision Badge & Toggle */}
                <div className="flex items-center gap-3 flex-shrink-0">
                  {/* Risk Score Pill */}
                  <div className="text-right hidden sm:block">
                    <div className="text-[10px] text-zinc-500 font-mono">RISK SCORE</div>
                    <div className={`font-bold font-mono text-sm ${
                      isCritical ? 'text-rose-400 animate-pulse' : ev.riskScore >= 0.3 ? 'text-amber-400' : 'text-emerald-400'
                    }`}>
                      {ev.riskScore.toFixed(2)}
                    </div>
                  </div>

                  <div>
                    {getDecisionBadge(ev.decision)}
                  </div>

                  <div className="text-zinc-500 hover:text-zinc-300">
                    {isExpanded ? <ChevronUp className="w-4 h-4" /> : <ChevronDown className="w-4 h-4" />}
                  </div>
                </div>
              </div>

              {/* Expanded Inspection Drawer */}
              {isExpanded && (
                <div className="mt-3 p-3 bg-zinc-900/80 border border-zinc-800 rounded-xl space-y-2 text-xs font-mono">
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-2 text-[11px]">
                    <div>
                      <span className="text-zinc-500 block">INTERCEPT REASON:</span>
                      <span className={`font-mono block mt-0.5 ${
                        isCritical ? 'text-rose-300' : 'text-zinc-300'
                      }`}>
                        {ev.reason}
                      </span>
                    </div>
                    
                    <div>
                      <span className="text-zinc-500 block">CHAMBER ISOLATION & LOCATION:</span>
                      <span className="text-cyan-300 font-mono block mt-0.5">
                        {ev.chamberIsolation || 'Chamber 02 (AI Quarantine Buffer)'}
                      </span>
                    </div>
                  </div>

                  <div className="grid grid-cols-1 sm:grid-cols-3 gap-2 pt-2 border-t border-zinc-800/80 text-[11px]">
                    <div>
                      <span className="text-zinc-500">ASSURANCE LEVEL:</span>
                      <span className="text-[#FACC15] ml-1 font-bold">{ev.ialAal}</span>
                    </div>
                    <div>
                      <span className="text-zinc-500">GAS / FINANCIAL:</span>
                      <span className="text-emerald-400 ml-1">{ev.financialImpact || 'N/A'}</span>
                    </div>
                    <div>
                      <span className="text-zinc-500">TIMESTAMP:</span>
                      <span className="text-zinc-400 ml-1">{ev.timestamp.replace('T', ' ').slice(0, 19)}</span>
                    </div>
                  </div>
                </div>
              )}
            </div>
          );
        })}
      </div>
    </div>
  );
};
