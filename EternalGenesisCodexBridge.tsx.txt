              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 font-mono text-xs">
                <div className="p-3 rounded-xl bg-zinc-950 border border-zinc-800 space-y-1">
                  <span className="text-zinc-500 text-[10px] uppercase">ผู้ประกาศิต / ผู้มีอำนาจออกกฎ (Authority)</span>
                  <div className="text-yellow-400 font-bold flex items-center gap-1.5">
                    <Award className="w-3.5 h-3.5" />
                    <span>{selectedEntry.authority}</span>
                  </div>
                </div>
                <div className="p-3 rounded-xl bg-zinc-950 border border-zinc-800 space-y-1">
                  <span className="text-zinc-500 text-[10px] uppercase">ประทับเวลาทางนิติวิทยาศาสตร์ (Forensic Time)</span>
                  <div className="text-cyan-300 font-bold">
                    {selectedEntry.timestamp}
                  </div>
                </div>
              </div>

              {/* Full Text Decree */}
              <div className="space-y-2">
                <div className="flex items-center gap-2 text-xs font-mono text-zinc-300 font-bold">
                  <Scroll className="w-4 h-4 text-yellow-400" />
                  <span>ตัวบทคำประกาศิต / กฎเกณฑ์ที่บังคับใช้ (Binding Decree Text)</span>
                </div>
                <div className="p-4 rounded-xl bg-black/70 border border-zinc-800/90 text-sm leading-relaxed text-zinc-200 font-serif">
                  &ldquo;{selectedEntry.fullDecree}&rdquo;
                </div>
              </div>

              {/* Legal Reference Badges */}
              <div className="space-y-2">
                <span className="text-[11px] font-mono text-zinc-400 font-bold">
                  บทบัญญัติทางกฎหมายและมาตรฐานที่อ้างอิง:
                </span>
                <div className="flex flex-wrap gap-2">
                  {selectedEntry.legalArticles.map((art, idx) => (
                    <span 
                      key={idx}
                      className="px-2.5 py-1 rounded-lg bg-zinc-800 border border-zinc-700 text-zinc-300 text-xs font-mono flex items-center gap-1"
                    >
                      <Scale className="w-3 h-3 text-yellow-400" />
                      <span>{art}</span>
                    </span>
                  ))}
                </div>
              </div>

              {/* Cryptographic Seal Box */}
              <div className="p-3.5 rounded-xl bg-zinc-950 border border-cyan-900/40 space-y-2 font-mono text-xs">
                <div className="flex items-center justify-between text-cyan-400 font-bold text-[11px]">
                  <span className="flex items-center gap-1.5">
                    <Fingerprint className="w-3.5 h-3.5" />
                    <span>Genesis Merkle Seal</span>
                  </span>
                  <span className="text-[10px] text-zinc-500">NIST PQC Verified</span>
                </div>
                <div className="p-2 bg-black rounded-lg border border-zinc-800 text-[11px] text-yellow-300/90 break-all select-all font-mono">
                  {selectedEntry.merkleSeal}
                </div>
                <div className="flex items-center justify-between text-[10px] text-zinc-400">
                  <span>Quantum Scheme: {selectedEntry.quantumHash}</span>
                  <span className="text-emerald-400">SSoT Validated &#10004;</span>
                </div>
              </div>
            </div>
          ) : (
            <div className="h-64 flex flex-col items-center justify-center text-zinc-500 font-mono text-xs border border-zinc-800 rounded-2xl">
              <BookOpen className="w-8 h-8 mb-2 opacity-40" />
              <span>เลือกข้อบัญญัติจากรายการด้านซ้ายเพื่อตรวจสอบ</span>
            </div>
          )}

          {/* Notarization Form: Sovereign Architect Decree Proclamation */}
          <form onSubmit={handleNotarizeDecree} className="p-4 rounded-2xl bg-zinc-900/60 border border-yellow-900/40 space-y-3">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-2 text-xs font-mono font-bold text-yellow-400">
                <Sparkles className="w-4 h-4" />
                <span>ตราประกาศิตใหม่ลงสู่ Eternal Genesis Codex</span>
              </div>
              <span className="text-[10px] font-mono text-zinc-400">
                Authorized: {SSOT.sovereignPrincipal}
              </span>
            </div>

            <input 
              type="text"
              placeholder="หัวข้อประกาศิต หรือ คำวินิจฉัยอธิปไตย..."
              value={newDecreeTitle}
              onChange={(e) => setNewDecreeTitle(e.target.value)}
              className="w-full bg-zinc-950 border border-zinc-800 rounded-xl px-3.5 py-2 text-xs font-mono text-zinc-200 placeholder-zinc-500 focus:outline-none focus:border-yellow-500"
            />

            <textarea 
              rows={2}
              placeholder="ระบุเนื้อหาคำสั่ง / ข้อตกลงที่จะประทับตรา Merkle และ Post-Quantum Seal..."
              value={newDecreeText}
              onChange={(e) => setNewDecreeText(e.target.value)}
              className="w-full bg-zinc-950 border border-zinc-800 rounded-xl p-3 text-xs font-mono text-zinc-200 placeholder-zinc-500 focus:outline-none focus:border-yellow-500"
            />

            <div className="flex items-center justify-between">
              {notarizedSuccess ? (
                <span className="text-xs font-mono text-emerald-400 flex items-center gap-1.5 font-bold">
                  <CheckCircle2 className="w-4 h-4" />
                  <span>ประทับตรา Merkle Root และบันทึกลง Codex สำเร็จ!</span>
                </span>
              ) : (
                <span className="text-[11px] font-mono text-zinc-500">
                  รองรับ ETDA Sec 9/26/28 และ PDPA 2019 อัตโนมัติ
                </span>
              )}

              <button
                type="submit"
                disabled={isNotarizing || !newDecreeTitle || !newDecreeText}
                className="px-4 py-2 bg-yellow-500 hover:bg-yellow-400 disabled:opacity-50 text-black font-bold font-mono text-xs rounded-xl transition flex items-center gap-2 cursor-pointer shadow-lg shadow-yellow-950/40"
              >
                {isNotarizing ? (
                  <>
                    <Cpu className="w-3.5 h-3.5 animate-spin" />
                    <span>กำลังประทับตรา PQC...</span>
                  </>
                ) : (
                  <>
                    <Lock className="w-3.5 h-3.5" />
                    <span>ประทับตรา Genesis Seal</span>
                  </>
                )}
              </button>
            </div>
          </form>
        </div>
      </div>
    </div>
  );
};
